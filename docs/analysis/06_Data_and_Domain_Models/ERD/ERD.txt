@startuml
entity User {
  +user_id : PK
  name
  email
  role <<Student | Teacher | Admin>>
  password_hash
  created_at
}

entity Assessment {
  +assessment_id : PK
  title
  description
  created_by : FK
  created_at
}

entity Question {
  +question_id : PK
  assessment_id : FK
  question_text
  max_marks
  created_at
}

entity ReferenceAnswer {
  +reference_id : PK
  question_id : FK
  answer_text
  embedding_vector
  created_at
}

entity Submission {
  +submission_id : PK
  question_id : FK
  student_id : FK
  answer_text
  embedding_vector
  submitted_at
}

entity Evaluation {
  +evaluation_id : PK
  submission_id : FK
  similarity_score
  marks_awarded
  feedback
  evaluated_at
}

entity AuditLog {
  +log_id : PK
  user_id : FK
  action
  entity_name
  entity_id
  timestamp
}

' Relationships
User ||--o{ Assessment : creates
Assessment ||--o{ Question : contains
Question ||--|| ReferenceAnswer : has
Question ||--o{ Submission : receives
User ||--o{ Submission : submits
Submission ||--|| Evaluation : evaluated_as
User ||--o{ AuditLog : generates

@enduml
